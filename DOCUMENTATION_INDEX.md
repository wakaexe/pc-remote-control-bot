## 📁 Структура проекта

```
pc_remote_bot/
├── 📄 Документация (вы здесь)
│   ├── README.md                              # Основная документация
│   ├── ИНСТРУКЦИЯ_ДЛЯ_ПОЛЬЗОВАТЕЛЕЙ.md        # Для пользователей
│   ├── READY_TO_DEPLOY.md                     # Готовность к деплою
│   ├── DEPLOYMENT.md                          # Инструкция по деплою
│   ├── GIT_COMMANDS.md                        # Git команды
│   └── ФИНАЛЬНЫЙ_СТАТУС.md                    # Финальный статус
│
├── 🤖 Боты
│   ├── main.py                                # Основной бот
│   ├── admin_bot.py                           # Админ бот
│   └── websocket_server.py                    # WebSocket сервер
│
├── 💻 Клиент
│   ├── pc_client.py                           # Клиент для ПК
│   ├── install_client.bat                     # Установщик Windows
│   └── install_client.sh                      # Установщик Linux/Mac
│
├── 📦 Модули
│   ├── bot/                                   # Telegram bot handlers
│   ├── pc/                                    # PC control modules
│   ├── utils/                                 # Utilities
│   ├── database.py                            # Database models
│   └── user_service.py                        # User management
│
└── ⚙️ Конфигурация
    ├── config.py                              # Настройки
    ├── requirements.txt                       # Зависимости
    ├── render.yaml                            # Render.com config
    └── Procfile                               # Процессы
```
